  27def _tutorial_contrast(image):
  28    """Return a display range without changing the measured image."""
  29    low, high = np.nanpercentile(image, (1, 99.7))
  30    return (low, high if high > low else low + 1)

  33def _tutorial_crop(shape, center, window_px):
  34    half = window_px // 2
  35    y = int(np.clip(center[0] - half, 0, max(shape[0] - window_px, 0)))
  36    x = int(np.clip(center[1] - half, 0, max(shape[1] - window_px, 0)))
  37    return slice(y, min(y + window_px, shape[0])), slice(x, min(x + window_px, shape[1]))

  40def pick_representative_cell(image_objects, margin_px=20):
  41    """Pick an interior, isolated, median-sized cell with valid geometry.
  42
  43    The rule is deterministic and uses geometry only, never fluorescence intensity.
  44    Returns the image object and cell ID so the same cell can be revisited.
  45    """
  46    candidates = []
  47    for image in image_objects:
  48        height, width = image.image.shape
  49        for cell in getattr(image, "cells", []):
  50            contour = np.asarray(cell.contour)
  51            mesh = np.asarray(cell.mesh)
  52            if (contour.ndim != 2 or contour.shape[1] != 2 or len(contour) < 6
  53                    or mesh.ndim != 2 or mesh.shape[1] != 4 or len(mesh) < 4
  54                    or not np.isfinite(contour).all() or not np.isfinite(mesh).all()):
  55                continue
  56            low = contour.min(axis=0)
  57            high = contour.max(axis=0)
  58            if (low[0] < margin_px or low[1] < margin_px
  59                    or high[0] > height - margin_px or high[1] > width - margin_px):
  60                continue
  61            box = high - low
  62            long_side, short_side = max(box), min(box)
  63            if short_side < 5 or long_side / short_side < 1.5:
  64                continue
  65            center = contour.mean(axis=0)
  66            center_distance = np.linalg.norm(center / [height, width] - 0.5)
  67            candidates.append((image, cell.cell_id, long_side, center_distance, center))
  68    if not candidates:
  69        raise ValueError("No interior cells with valid elongated contours and meshes")
  70    median_length = np.median([item[2] for item in candidates])
  71    typical = [item for item in candidates if abs(item[2] - median_length) <= 0.25 * median_length]
  72    neighbors = {}
  73    for image in image_objects:
  74        group = [item for item in candidates if item[0] is image]
  75        if not group:
  76            continue
  77        all_centers = [np.asarray(cell.contour).mean(axis=0)
  78                       for cell in image.cells
  79                       if np.asarray(cell.contour).ndim == 2
  80                       and np.asarray(cell.contour).shape[1] == 2
  81                       and len(cell.contour) > 0]
  82        if len(all_centers) < 2:
  83            for item in group:
  84                neighbors[(image.image_name, item[1])] = np.inf
  85            continue
  86        distances = cKDTree(all_centers).query([item[4] for item in group], k=2)[0]
  87        for item, distance in zip(group, distances):
  88            neighbors[(image.image_name, item[1])] = distance[1] / item[2]
  89    winner = min(
  90        typical,
  91        key=lambda item: (-neighbors[(item[0].image_name, item[1])],
  92                          item[3], abs(item[2] - median_length),
  93                          item[0].image_name, item[1]),
  94    )
  95    return winner[0], winner[1]

  98def select_example_cell(image_objects, field_id=None, cell_id=None):
  99    """Choose an existing field/cell, or a representative cell in the chosen field."""
 100    objects = list(image_objects)
 101    if field_id is not None:
 102        objects = [image for image in objects if image.image_name.rsplit("_", 1)[0] == field_id]
 103        if not objects:
 104            raise ValueError(f"Field {field_id!r} was not found")
 105    if cell_id is None:
 106        return pick_representative_cell(objects)
 107    if field_id is None:
 108        raise ValueError("Set field_id when selecting a cell_id")
 109    for image in objects:
 110        if any(cell.cell_id == cell_id for cell in image.cells):
 111            return image, cell_id
 112    raise ValueError(f"Cell {cell_id} was not found in field {field_id}")

 115def _tutorial_channel_color(label):

 160def plot_segmentation_review(image_objects, crop_size=500, mask_label=None, center=None):
 161    """Compare phase and mask outlines in a selectable cell-centered crop."""
 162    objects = list(image_objects)
 163    if not objects:
 164        raise ValueError("image_objects is empty")
 165    fig, axes = plt.subplots(len(objects), 2, figsize=(10, 4.5 * len(objects)), squeeze=False)
 166    for row, image in enumerate(objects):
 167        if image.mask is None:
 168            raise ValueError(f"No mask is loaded for {image.image_name}")
 169        labels = np.unique(image.mask)
 170        labels = labels[labels > 0]
 171        if mask_label is not None and mask_label not in labels:
 172            raise ValueError(f"Mask label {mask_label} was not found in {image.image_name}")
 173        chosen = mask_label
 174        if chosen is None and center is None and len(labels):
 175            midpoint = np.asarray(image.image.shape) / 2
 176            nonzero = np.argwhere(image.mask > 0)
 177            closest = np.argmin(np.sum((nonzero - midpoint) ** 2, axis=1))
 178            chosen = image.mask[tuple(nonzero[closest])]
 179        if chosen is not None:
 180            focus = np.argwhere(image.mask == chosen).mean(axis=0)
 181        else:
 182            focus = center if center is not None else np.asarray(image.image.shape) / 2
 183        crop = _tutorial_crop(image.image.shape, focus, crop_size)
 184        phase, mask = image.image[crop], image.mask[crop]
 185        low, high = _tutorial_contrast(phase)
 186        for ax in axes[row]:
 187            ax.imshow(phase, cmap="gray", vmin=low, vmax=high)
 188            ax.set_xticks([])
 189            ax.set_yticks([])
 190            for spine in ax.spines.values():
 191                spine.set_visible(False)
 192        outlines = np.ma.masked_where(~find_boundaries(mask, mode="outer"), np.ones(mask.shape))
 193        axes[row, 1].imshow(outlines, cmap="autumn", vmin=0, vmax=1, alpha=0.9, interpolation="nearest")
 194        if chosen is not None:
 195            selected_outline = find_boundaries(mask == chosen, mode="outer")
 196            axes[row, 1].imshow(np.ma.masked_where(~selected_outline, np.ones(mask.shape)),
 197                                cmap="autumn", vmin=0, vmax=1,
 198                                interpolation="nearest")
 199        field = image.image_name.rsplit("_", 1)[0]
 200        axes[row, 0].set_title(f"{field}  |  phase", loc="left", fontsize=11)
 201        axes[row, 1].set_title(f"Mask outlines  |  selected label {chosen}", loc="left", fontsize=11)
 202    fig.suptitle("SEGMENTATION REVIEW", x=0.04, y=0.98, ha="left", fontsize=15, fontweight="bold", color="#193348")
 203    fig.tight_layout(rect=(0, 0, 1, 0.91))
 204    return fig

 207def plot_cell_gallery(image_object, cell_id, channel="C2", window_px=110,
 208                      channel_label=None, align_signal=False, show_objects=False):
 209    """Show phase and mesh, optionally with a registered signal crop and objects.
 210
 211    The registered crop uses the same local crop/phase registration called by
 212    ``object_detection(align=True)``. Plot it after detection to compare its
 213    detected object contours with the phase-cell contour.
 214    """
 215    cell = next((item for item in image_object.cells if item.cell_id == cell_id), None)
 216    if cell is None:
 217        raise ValueError(f"Cell {cell_id} was not found")
 218    if channel is not None and (not image_object.channels or channel not in image_object.channels):
 219        raise ValueError(f"Channel {channel} is not loaded")
 220    contour = np.asarray(cell.contour)
 221    mesh = np.asarray(cell.mesh)
 222    midline = np.asarray(cell.midline)
 223    extent = int(np.ceil(np.max(np.ptp(contour, axis=0)))) + 24
 224    crop = _tutorial_crop(image_object.image.shape, contour.mean(axis=0), max(window_px, extent))
 225    phase = image_object.image[crop]
 226    label = channel_label or channel
 227    panels = [(phase, "gray", "01  PHASE"), (phase, "gray", "02  CONTOUR + MESH")]
 228    signal_origin = (crop[1].start, crop[0].start)
 229    if channel is not None:
 230        signal = image_object.channels[channel][crop]
 231        if align_signal:
 232            signal, _, _, row_offset, column_offset = u.crop_image(
 233                image_object.channels[channel], cell.contour,
 234                phase_img=image_object.image,
 235            )
 236            signal_origin = (column_offset, row_offset)
 237        heading = "ALIGNED " if align_signal else ""
 238        panels.append((signal, _tutorial_channel_cmap(label),
 239                       f"03  {heading}{channel} {label.upper()}"))
 240    fig, axes = plt.subplots(1, len(panels), figsize=(3.7 * len(panels), 3.9))
 241    for ax, (data, cmap, title) in zip(axes, panels):
 242        low, high = _tutorial_contrast(data)
 243        ax.imshow(data, cmap=cmap, vmin=low, vmax=high)
 244        ax.set_title(title, loc="left", fontsize=10, fontweight="bold", color="#193348")
 245        ax.set_xticks([])
 246        ax.set_yticks([])
 247        for spine in ax.spines.values():
 248            spine.set_visible(False)
 249    x0, y0 = crop[1].start, crop[0].start
 250    axes[1].plot(contour[:, 1] - x0, contour[:, 0] - y0, color="#FFCC66", lw=1.6)
 251    if channel is not None:
 252        axes[2].plot(contour[:, 1] - signal_origin[0], contour[:, 0] - signal_origin[1], color="#FFCC66", lw=1.6)
 253        if show_objects:
 254            for detected in cell.object_meshdata.get(channel, {}).get("object_contour", []):
 255                detected = np.asarray(detected)
 256                if detected.ndim == 2 and detected.shape[1] == 2:
 257                    axes[2].plot(detected[:, 1] - signal_origin[0],
 258                                 detected[:, 0] - signal_origin[1],
 259                                 color="#4DE6F2", lw=1.2)
 260    if mesh.ndim == 2 and mesh.shape[1] == 4:
 261        axes[1].plot(mesh[:, 1] - x0, mesh[:, 0] - y0, color="#36D6D0", lw=0.7)
 262        axes[1].plot(mesh[:, 3] - x0, mesh[:, 2] - y0, color="#36D6D0", lw=0.7)
 263        for segment in mesh[::3]:
 264            axes[1].plot([segment[1] - x0, segment[3] - x0], [segment[0] - y0, segment[2] - y0], color="#36D6D0", lw=0.55)
 265    if midline.ndim == 2 and midline.shape[1] == 2:
 266        axes[1].plot(midline[:, 1] - x0, midline[:, 0] - y0, color="#F26463", lw=1.3)
 267    max_scale_px = phase.shape[1] * 0.45
 268    scale_um = next(
 269        (value for value in (5, 2, 1, 0.5, 0.2, 0.1)
 270         if value / image_object.px <= max_scale_px),
 271        0.1,
 272    )
 273    scale_px = max(1, round(scale_um / image_object.px))
 274    axes[0].plot([7, 7 + scale_px], [phase.shape[0] - 9] * 2, color="white", lw=3, solid_capstyle="butt")
 275    axes[0].text(7, phase.shape[0] - 14, f"{scale_um:g} µm", color="white", fontsize=8)
 276    fig.suptitle(f"CELL {cell_id}  /  {image_object.image_name}", x=0.02, y=0.98, ha="left", fontsize=13, fontweight="bold", color="#193348")
 277    fig.tight_layout(rect=(0, 0, 1, 0.93))
 278    return fig

 280def plot_C2_alignment(image_object, cell_id, window_px=110,
 281                      channel="C2", channel_label=None,
 282                      show_objects=False):
 283    """
 284    Compare the raw C2 signal with the phase-aligned C2 signal.
 285
 286    BEFORE: original fluorescence image, cropped around the cell.
 287    AFTER:  fluorescence crop returned by u.crop_image(..., phase_img=...).
 288
 289    Cell contours are overlaid using the correct origin for each crop.
 290    Detected object contours can optionally be shown on the aligned panel.
 291    """
 292
 293    cell = next(
 294        (item for item in image_object.cells if item.cell_id == cell_id),
 295        None,
 296    )
 297    if cell is None:
 298        raise ValueError(f"Cell {cell_id} was not found")
 299
 300    if not image_object.channels or channel not in image_object.channels:
 301        raise ValueError(f"Channel {channel} is not loaded")
 302
 303    contour = np.asarray(cell.contour)
 304    label = channel_label or channel
 305
 306    # BEFORE: use the original image and the same cropping method
 307    # as plot_cell_gallery().
 308    extent = int(np.ceil(np.max(np.ptp(contour, axis=0)))) + 24
 309
 310    crop = _tutorial_crop(
 311        image_object.image.shape,
 312        contour.mean(axis=0),
 313        max(window_px, extent),
 314    )
 315
 316    signal_before = image_object.channels[channel][crop]
 317
 318    # Crop origin in image coordinates: (column, row).
 319    origin_before = (
 320        crop[1].start,
 321        crop[0].start,
 322    )
 323
 324    # AFTER: apply the same local registration as in plot_cell_gallery().
 325    signal_after, _, _, row_offset, column_offset = u.crop_image(
 326        image_object.channels[channel],
 327        cell.contour,
 328        phase_img=image_object.image,
 329    )
 330
 331    # The function returns row offset first, column offset second.
 332    # For plotting, x = column and y = row.
 333    origin_after = (
 334        column_offset,
 335        row_offset,
 336    )
 337
 338    panels = [
 339        (signal_before, origin_before, f"01  UNALIGNED {channel}"),
 340        (signal_after, origin_after, f"02  ALIGNED {channel}"),
 341    ]
 342
 343    fig, axes = plt.subplots(
 344        1, 2,
 345        figsize=(10, 4.8),
 346    )
 347
 348    # Use identical contrast limits in both panels.
 349    combined = np.concatenate([
 350        signal_before.ravel(),
 351        signal_after.ravel(),
 352    ])
 353    combined = combined[np.isfinite(combined)]
 354
 355    if combined.size == 0:
 356        raise ValueError("Both signal crops contain no finite values")
 357
 358    low, high = np.percentile(combined, [1, 99.5])
 359
 360    if low == high:
 361        high = low + 1
 362
 363    for ax, (signal, origin, title) in zip(axes, panels):
 364
 365        ax.imshow(
 366            signal,
 367            cmap=_tutorial_channel_cmap(label),
 368            vmin=low,
 369            vmax=high,
 370        )
 371
 372        ax.set_title(
 373            title,
 374            loc="left",
 375            fontsize=11,
 376            fontweight="bold",
 377            color="#193348",
 378        )
 379
 380        ax.set_xticks([])
 381        ax.set_yticks([])
 382
 383        for spine in ax.spines.values():
 384            spine.set_visible(False)
 385
 386        # Cell contours use (row, column).
 387        # Matplotlib uses (x=column, y=row).
 388        x0, y0 = origin
 389
 390        ax.plot(
 391            contour[:, 1] - x0,
 392            contour[:, 0] - y0,
 393            color="#FFCC66",
 394            lw=1.6,
 395        )
 396
 397    # Show detected objects only on the aligned panel.
 398    # These contours are taken directly from the cell object.
 399    if show_objects:
 400        x0, y0 = origin_after
 401
 402        for detected in cell.object_meshdata.get(
 403            channel, {}
 404        ).get("object_contour", []):
 405
 406            detected = np.asarray(detected)
 407
 408            if detected.ndim == 2 and detected.shape[1] == 2:
 409                axes[1].plot(
 410                    detected[:, 1] - x0,
 411                    detected[:, 0] - y0,
 412                    color="#4DE6F2",
 413                    lw=1.2,
 414                )
 415
 416    fig.suptitle(
 417        f"{channel} ALIGNMENT  /  CELL {cell_id}  /  "
 418        f"{image_object.image_name}",
 419        x=0.02,
 420        y=0.98,
 421        ha="left",
 422        fontsize=13,
 423        fontweight="bold",
 424        color="#193348",
 425    )
 426
 427    fig.tight_layout(rect=(0, 0, 1, 0.93))
 428
 429    return fig

 440def plot_curation_review(before_contours, image_objects, crop_size=700):
 441    """Show retained cells in green and removed cells in red on phase images.
 442
 443    ``before_contours`` is made by :func:`snapshot_cell_contours` before
 444    curation. Cells are matched by image name and cell ID.
 445    """
 446    objects = list(image_objects)
 447    if not objects:
 448        raise ValueError("image_objects is empty")
 449    fig, axes = plt.subplots(1, len(objects), figsize=(5.2 * len(objects), 5.2), squeeze=False)
 450    for ax, image in zip(axes[0], objects):
 451        original = before_contours.get(image.image_name)
 452        if original is None:
 453            raise ValueError(f"No pre-curation contours for {image.image_name}")
 454        kept = {cell.cell_id for cell in image.cells}
 455        if not kept.issubset(original):
 456            raise ValueError(f"Curated cell IDs do not match the original meshes for {image.image_name}")
 457        for cell in image.cells:
 458            old_center = np.asarray(original[cell.cell_id]).mean(axis=0)
 459            new_center = np.asarray(cell.contour).mean(axis=0)
 460            if np.linalg.norm(old_center - new_center) > 1:
 461                raise ValueError(f"Curated cell geometry does not match the original mesh for {image.image_name}")
 462        crop = _tutorial_crop(
 463            image.image.shape,
 464            (image.image.shape[0] // 2, image.image.shape[1] // 2),
 465            crop_size,
 466        )
 467        phase = image.image[crop]
 468        low, high = _tutorial_contrast(phase)
 469        ax.imshow(phase, cmap="gray", vmin=low, vmax=high)
 470        x0, y0 = crop[1].start, crop[0].start
 471        for cell_id, contour in original.items():
 472            contour = np.asarray(contour)
 473            margin = 3
 474            fully_inside = (
 475                (contour[:, 0] >= crop[0].start + margin)
 476                & (contour[:, 0] < crop[0].stop - margin)
 477                & (contour[:, 1] >= crop[1].start + margin)
 478                & (contour[:, 1] < crop[1].stop - margin)
 479            )
 480            if not fully_inside.all():
 481                continue
 482            keep = cell_id in kept
 483            ax.plot(
 484                contour[:, 1] - x0, contour[:, 0] - y0,
 485                color="#34D986" if keep else "#FF686B",
 486                lw=1.15 if keep else 0.85,
 487                alpha=0.95 if keep else 0.82,
 488            )
 489        field = image.image_name.rsplit("_", 1)[0]
 490        ax.set_title(
 491            f"{field}\n{len(kept):,} kept  ·  {len(original) - len(kept):,} removed",
 492            loc="left", fontsize=11, color="#193348",
 493        )
 494        ax.set_xticks([])
 495        ax.set_yticks([])
 496        for spine in ax.spines.values():
 497            spine.set_visible(False)
 498    fig.legend(
 499        handles=[
 500            Line2D([], [], color="#34D986", lw=2, label="Kept"),
 501            Line2D([], [], color="#FF686B", lw=2, label="Removed"),
 502        ],
