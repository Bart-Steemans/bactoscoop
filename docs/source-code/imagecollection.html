  33        self, image_folder_path=None, px=0.065, error_policy="continue",

  38        ``error_policy="continue"`` records item-level errors and proceeds with
  39        independent fields. ``error_policy="raise"`` raises on the first one.

  45            raise ValueError("error_policy must be either 'continue' or 'raise'.")

  47        self.log_cell_errors = log_cell_errors

  49        self.cell_errors = []
  50        self.completed_stages = set()

  54        self.masks = None
  55        self.channel_images = {}
  56        self.channel_image_filenames = {}
  57        self.channel_images_by_field = {}
  58        self.inverted_image = None
  59        self.channel_list = None
  60        self.image_filenames = None
  61        self.mask_filenames = None
  62        self.phase_channel = ""
  63        self.segmentation_failed = False
  64
  65        self.image_folder_path = image_folder_path  # Path to the folder to be processed
  66        self.name = (
  67            os.path.basename(os.path.normpath(self.image_folder_path))

  91        self, stage, error, image_name=None, channel=None, cell_id=None

  93        record = {
  94            "stage": stage,
  95            "image_name": image_name,
  96            "channel": channel,
  97            "cell_id": cell_id,
  98            "error_type": type(error).__name__,
  99            "error": str(error),

 101        self.processing_errors.append(record)
 102        bactoscoop_logger.error(

 108            error,
 109        )

 135    def processing_summary(self):
 136        """Return attempted-stage status and retained image/cell error records."""
 137        if self.processing_errors:
 138            status = "completed_with_errors"
 139        elif self.completed_stages:
 140            status = "completed"
 141        else:
 142            status = "not_started"

 145            "completed_stages": sorted(self.completed_stages),
 146            "error_count": len(self.processing_errors),
 147            "errors": list(self.processing_errors),
 148            "cell_error_count": len(self.cell_errors),
 149            "cell_errors": list(self.cell_errors),

 155    def _validate_phase_image_state(self):
 156        if self.images is None or self.image_filenames is None:
 157            raise ValueError(
 158                "Phase images are not loaded. Run load_phase_images() before creating image objects."
 159            )
 160        if self.masks is None:
 161            raise ValueError(
 162                "Masks are not loaded. Run load_masks() before creating image objects."
 163            )
 164        if len(self.image_filenames) != len(self.images):
 165            raise ValueError(
 166                f"Loaded {len(self.image_filenames)} image names but {len(self.images)} phase images."
 167            )
 168        if self.mask_filenames is None or len(self.mask_filenames) != len(self.masks):
 169            raise ValueError("Mask arrays and mask filenames are not aligned.")
 170
 171        phase_by_key = {}
 172        for index, filename in enumerate(self.image_filenames):
 173            key = u.image_field_key(filename, suffix=self.phase_channel)
 174            if key in phase_by_key:
 175                self._record_processing_error(
 176                    "phase_mask_matching",
 177                    ValueError(f"Duplicate phase field identity '{key}'."),
 178                    image_name=filename,
 179                )
 180                continue
 181            phase_by_key[key] = (index, filename, self.images[index])
 182
 183        masks_by_key = {}
 184        for index, filename in enumerate(self.mask_filenames):
 185            key = u.image_field_key(
 186                filename, suffix=self.phase_channel, is_mask=True
 187            )
 188            if key in masks_by_key:
 189                self._record_processing_error(
 190                    "phase_mask_matching",
 191                    ValueError(f"Duplicate mask field identity '{key}'."),
 192                    image_name=filename,
 193                )
 194                continue
 195            masks_by_key[key] = (filename, self.masks[index])
 196
 197        matched = []
 198        for key, (frame, image_name, image) in phase_by_key.items():
 199            mask_entry = masks_by_key.get(key)
 200            if mask_entry is None:
 201                self._record_processing_error(
 202                    "phase_mask_matching",
 203                    ValueError(f"No mask matches phase field '{image_name}'."),
 204                    image_name=image_name,
 205                )
 206                continue
 207            mask_name, mask = mask_entry
 208            if image.ndim != 2 or mask.ndim != 2:
 209                self._record_processing_error(
 210                    "phase_mask_matching",
 211                    ValueError(
 212                        f"Phase and mask images must be 2D; got phase shape "
 213                        f"{image.shape} and mask '{mask_name}' shape {mask.shape}."
 214                    ),
 215                    image_name=image_name,
 216                )
 217                continue
 218            if image.shape != mask.shape:
 219                self._record_processing_error(
 220                    "phase_mask_matching",
 221                    ValueError(
 222                        f"Phase shape {image.shape} does not match mask "
 223                        f"'{mask_name}' shape {mask.shape}."
 224                    ),
 225                    image_name=image_name,
 226                )
 227                continue
 228            matched.append((frame, image_name, image, mask))
 229
 230        for key, (mask_name, _) in masks_by_key.items():
 231            if key not in phase_by_key:
 232                self._record_processing_error(
 233                    "phase_mask_matching",
 234                    ValueError(f"No phase image matches mask '{mask_name}'."),
 235                    image_name=mask_name,
 236                )
 237        if not matched:
 238            raise ValueError("No phase images have matching masks and dimensions.")
 239        return matched
 240    def _normalize_channel_list(self, channel_list, argument_name="channel_list"):
 241        if channel_list is None:
 242            raise ValueError(
 243                f"No {argument_name} provided. It should be of format ['channelname1', 'channelname2']."
 244            )
 245        if isinstance(channel_list, str):
 246            raise TypeError(
 247                f"{argument_name} must be an iterable of channel names, not a single string."
 248            )
 249        normalized_channels = list(channel_list)
 250        if not normalized_channels:
 251            raise ValueError(f"{argument_name} must contain at least one channel name.")
 252        return normalized_channels

 254    def _iter_processing_targets(self, object_list, phase_channel):
 255        if isinstance(object_list, Image):
 256            return [object_list]
 257        if object_list is None:

 262        raise TypeError(

 266    def _get_mesh_dataframe_lookup(self):
 267        """Build an image-name lookup from the current mesh dataframe.
 268
 269        Do not retain this lookup across calls: callers can replace or mutate
 270        ``mesh_df_collection``, and a cached lookup could then return stale
 271        meshes for newly loaded data.
 272        """
 273        if not isinstance(self.mesh_df_collection, pd.DataFrame) or self.mesh_df_collection.empty:
 274            return None
 275        return {
 276            image_name: group.reset_index(drop=True)
 277            for image_name, group in self.mesh_df_collection.groupby(
 278                "image_name", sort=False
 279            )
 280        }

 296    def load_phase_images(self, phase_channel=""):
 297        """
 298        This method loads phase contrast images from the specified image folder path, based on the characters before the file-extension,
 299        and stores them in the 'self.images' array.
 300
 301        Parameters
 302        ----------
 303        phase_channel : str, optional
 304            Characters before the file-extension (e.g., "c1", "phase") to look for when using this function. (default is an empty string)
 305
 306        Mutates ``self.images``, ``self.image_filenames`` and ``self.paths``.
 307
 308        """
 309        self._require_image_folder_path()
 310        self.phase_channel = phase_channel or ""
 311        try:
 312            self.images, self.image_filenames, self.paths = u.read_tiff_folder(
 313                self.image_folder_path, phase_channel, include_paths=True
 314            )
 315        except Exception as e:
 316            raise type(e)(
 317                f"Failed to load phase contrast images from '{self.image_folder_path}' "
 318                f"with phase_channel='{phase_channel}': {e}"
 319            ) from e
 320        self._record_stage_success("load_phase_images")

 322    def load_channel_images(self, channel_list):
 323        """
 324        This method loads channel images from the image folder path based on the specified characters before the file-extension and stores them in
 325        the 'self.channel_images' dictionary, where the channel name is used as the key.
 326
 327        Parameters
 328        ----------
 329        channel_list : List of str
 330            Characters before the extension (e.g., ["c2", "c3"]) to look for when using this function.
 331
 332        Mutates ``self.channel_images``.
 333
 334        """
 335        self._require_image_folder_path()
 336        self.channel_list = self._normalize_channel_list(channel_list)
 337        for channel in self.channel_list:
 338            # A failed read must not leave arrays from an earlier folder/load in place.
 339            self.channel_images.pop(channel, None)
 340            self.channel_image_filenames.pop(channel, None)
 341            self.channel_images_by_field.pop(channel, None)
 342            try:
 343                images, filenames = u.read_tiff_folder(
 344                    self.image_folder_path, suffix=channel
 345                )
 346            except Exception as e:
 347                self._record_processing_error(
 348                    "load_channel_images", e, channel=channel
 349                )
 350                continue
 351
 352            self.channel_images[channel] = list(images)
 353            self.channel_image_filenames[channel] = list(filenames)
 354            keyed_images = {}
 355            for filename, image in zip(filenames, images):
 356                key = u.image_field_key(filename, suffix=channel)
 357                if key in keyed_images:
 358                    self._record_processing_error(
 359                        "load_channel_images",
 360                        ValueError(f"Duplicate channel field identity '{key}'."),
 361                        image_name=filename,
 362                        channel=channel,
 363                    )
 364                    continue
 365                keyed_images[key] = image
 366            self.channel_images_by_field[channel] = keyed_images
 367        self._record_stage_success("load_channel_images")

 369    def create_image_objects(self, phase_channel=""):
 370        """
 371        Create ``Image`` objects from the loaded masks and phase images.
 372
 373        Invalid collection-level setup raises immediately. Failures while creating
 374        a single image object are logged and skipped so the rest of the batch can continue.
 375        """
 376        if self.segmentation_failed:
 377            raise RuntimeError(
 378                "Segmentation failed in this collection; refusing to load masks "
 379                "from disk because they may belong to an earlier run."
 380            )
 381        if self.masks is None:
 382            self.load_masks()
 383        if self.images is None or self.image_filenames is None:
 384            self.load_phase_images(phase_channel)
 385
 386        matched_images = self._validate_phase_image_state()
 387
 388        # Feature tables and merged output belong to the previous image/cell set.
 389        self.feature_dataframes = {}
 390        self.merged_features = None
 391        self.image_objects = []
 392        mesh_lookup = self._get_mesh_dataframe_lookup()
 393
 394        for i, image_name, image, mask in matched_images:
 395            try:
 396                mesh_df = mesh_lookup.get(image_name) if mesh_lookup is not None else None
 397
 398                img_obj = self.create_image_object(
 399                    image, image_name, i, mask, mesh_df, self.px
 400                )
 401                self.image_objects.append(img_obj)
 402            except Exception as e:
 403                bactoscoop_logger.debug(traceback.format_exc())
 404                self._record_processing_error(
 405                    "create_image_objects", e, image_name=image_name
 406                )
 407        self._record_stage_success("create_image_objects")

 409    def add_channels(self, img_objects, channel_list, load_data=True):
 410        """
 411        This method loads channel images based on the provided channel list and adds them to the individual image objects.
 412
 413        Parameters
 414        ----------
 415        img_objects : List
 416            List of image objects.
 417        channel_list : List
 418            List of channel names.
 419
 420        """
 421        channel_list = self._normalize_channel_list(channel_list)
 422        if img_objects is None or len(img_objects) == 0:
 423            raise ValueError("img_objects must contain at least one image object.")
 424
 425        # Load requested channels while keeping field identity beside each array.
 426        if load_data:
 427            self.load_channel_images(channel_list)
 428
 429        expected_fields = {
 430            u.image_field_key(obj.image_name, suffix=self.phase_channel)
 431            for obj in img_objects
 432        }
 433        for channel in channel_list:
 434            for extra_field in self.channel_images_by_field.get(channel, {}):
 435                if extra_field not in expected_fields:
 436                    self._record_processing_error(
 437                        "match_channel_image",
 438                        ValueError(
 439                            f"No phase image object matches {channel} field '{extra_field}'."
 440                        ),
 441                        image_name=extra_field,
 442                        channel=channel,
 443                    )
 444
 445        for img_obj in img_objects:
 446            channel_single_image_dict = dict(getattr(img_obj, "channels", None) or {})
 447            field_key = u.image_field_key(
 448                img_obj.image_name, suffix=self.phase_channel
 449            )
 450            for channel in channel_list:
 451                # Never retain an earlier image or derived interpolation for a
 452                # requested channel if its current file is missing or invalid.
 453                channel_single_image_dict.pop(channel, None)
 454                for cache_name in ("bg_channels", "chann_interp2d"):
 455                    cache = getattr(img_obj, cache_name, None)
 456                    if isinstance(cache, dict):
 457                        cache.pop(channel, None)
 458                channel_fields = self.channel_images_by_field.get(channel, {})
 459                channel_image = channel_fields.get(field_key)
 460                if channel_image is None:
 461                    self._record_processing_error(
 462                        "match_channel_image",
 463                        ValueError(
 464                            f"No {channel} image matches field '{img_obj.image_name}'."
 465                        ),
 466                        image_name=img_obj.image_name,
 467                        channel=channel,
 468                    )
 469                    continue
 470                if channel_image.ndim != 2 or img_obj.image.ndim != 2:
 471                    self._record_processing_error(
 472                        "match_channel_image",
 473                        ValueError(
 474                            f"{channel} and phase images must be 2D; got channel "
 475                            f"shape {channel_image.shape} and phase shape {img_obj.image.shape}."
 476                        ),
 477                        image_name=img_obj.image_name,
 478                        channel=channel,
 479                    )
 480                    continue
 481                if channel_image.shape != img_obj.image.shape:
 482                    self._record_processing_error(
 483                        "match_channel_image",
 484                        ValueError(
 485                            f"{channel} image shape {channel_image.shape} does not "
 486                            f"match phase shape {img_obj.image.shape}."
 487                        ),
 488                        image_name=img_obj.image_name,
 489                        channel=channel,
 490                    )
 491                    continue
 492                channel_single_image_dict[channel] = channel_image
 493            img_obj.channels = channel_single_image_dict or None
 494        self._record_stage_success("add_channels")

 512    def create_image_object(self, image, image_name, index, mask, mesh_df, px):
 513        """
 514        This method creates a single Image object using the provided image, image name, index, mask, mesh dataframe, and pixel size (px).
 515        If mesh data is present, it also creates associated Cell objects.
 516
 517        Parameters
 518        ----------
 519        image : phase contrast image
 520            The phase contrast image for the Image object.
 521        image_name : str
 522            The name or identifier of the image.
 523        index : int
 524            The index or position of the image.
 525        mask : mask or None
 526            The mask or region of interest associated with the image.
 527        mesh_df : pd.DataFrame or None

 540    def segment_images(self, mask_thresh, minsize, n, model_name="bact_phase_omni"):
 541        self.segmentation_failed = False
 542        self.masks = None
 543        self.mask_filenames = None
 544        try:
 545            from .omni import Omnipose
 546            import torch
 547
 548            omni = Omnipose(self.images, self.paths)
 549            omni.load_models(model_name)
 550
 551            omni.compiled_process(n, mask_thresh, minsize)
 552            torch.cuda.empty_cache()
 553
 554            del omni
 555            gc.collect()
 556            self._record_stage_success("segment_images")
 557            return True
 558        except Exception as e:
 559            self.segmentation_failed = True
 560            bactoscoop_logger.debug(traceback.format_exc())
 561            self._record_processing_error("segment_images", e)
 562            return False

 564    def batch_detect_objects(
 565        self,
 566        channels=None,
 567        reset_channels=True,
 568        align=False,
 569        smoothing=0.1,
 570        log_sigma=3,
 571        kernel_width=4,
 572        min_overlap_ratio=0.01,
 573        max_external_ratio=0.1,
 574    ):
 575        """
 576        Detect objects within the contour of the cell.
 577
 578        Parameters
 579        ----------
 580        channels : List of strings, optional
 581            The channel names of the channels to perform object detection on.
 582
 583        log_sigma : int, optional
 584            The sigma parameter of the Laplacian of Gaussian filter used in object detection. (default is 3)
 585
 586        kernel_width : int, optional
 587            The kernel width of the kernel used for dilating the object masks.
 588            Decreasing results in smaller masks, while increasing results in larger masks. (default is 4)
 589
 590        min_overlap_ratio : float, optional
 591            Minimum overlap between the object and cell required for the object to be kept. (default is 0.01)
 592
 593        max_external_ratio : float, optional
 594            The maximum ratio an object is allowed to lie outside the cell contour. (default is 0.1)
 595
 596        Returns
 597        -------
 598        pd.DataFrame
 599            A DataFrame containing cell_id, object_contours, and frame information.
 600
 601        """
 602        self._require_loaded_images()
 603        channels = self._normalize_channel_list(channels, argument_name="channels")
 604
 605        if reset_channels:
 606            self.channel_images = {}
 607            self.channel_image_filenames = {}
 608            self.channel_images_by_field = {}
 609            for image in self.image_objects:
 610                image.channels = None
 611                for cache_name in ("bg_channels", "chann_interp2d"):
 612                    cache = getattr(image, cache_name, None)
 613                    if isinstance(cache, dict):
 614                        cache.clear()
 615
 616        channels_to_load = [
 617            channel for channel in channels
 618            if channel not in self.channel_images_by_field
 619        ]
 620        if channels_to_load:
 621            self.load_channel_images(channels_to_load)
 622        self.add_channels(self.image_objects, channels, load_data=False)
 623
 624        bactoscoop_logger.info("Detecting objects within cell ...")
 625
 626        dfs = []
 627        for image in tqdm_if_verbose(self.image_objects):
 628            for cell in getattr(image, "cells", []):
 629                for channel in channels:
 630                    cell.object_meshdata.pop(channel, None)
 631            if image.channels is None or any(
 632                channel not in image.channels for channel in channels
 633            ):
 634                # The missing field has already been recorded by add_channels.
 635                continue
 636            try:
 637                detection_df = image.object_detection(
 638                    channels,
 639                    smoothing,
 640                    align,
 641                    log_sigma,
 642                    kernel_width,
 643                    min_overlap_ratio,
 644                    max_external_ratio,
 645                )
 646            except Exception as e:
 647                bactoscoop_logger.debug(traceback.format_exc())
 648                self._record_processing_error(
 649                    "object_detection", e, image_name=image.image_name
 650                )
 651                continue
 652            dfs.append(detection_df)
 653            cell_errors = getattr(image, "object_detection_errors", [])
 654            for cell_error in cell_errors:
 655                self._record_cell_error(
 656                    "cell_object_detection",
 657                    RuntimeError(cell_error["error"]),
 658                    image_name=image.image_name,
 659                    channel=cell_error.get("channel"),
 660                    cell_id=cell_error.get("cell_id"),
 661                    error_type=cell_error.get("error_type"),
 662                )
 663            cell_count = len(getattr(image, "cells", []))
 664            failed_attempts = sum(
 665                not error.get("recovered", False) for error in cell_errors
 666            )
 667            if cell_count and failed_attempts >= cell_count * len(channels):
 668                self._record_processing_error(
 669                    "object_detection",
 670                    ValueError(
 671                        f"All cell object-detection attempts failed for '{image.image_name}'."
 672                    ),
 673                    image_name=image.image_name,
 674                )
 675
 676        self.object_detection_df = (
 677            pd.concat(dfs, axis=0, ignore_index=True) if dfs else pd.DataFrame()
 678        )
 679
 680        if not self.object_detection_df.empty and "channel" in self.object_detection_df:
 681            bactoscoop_logger.info(self.object_detection_df["channel"].value_counts())
 682        else:
 683            bactoscoop_logger.info("No objects were detected.")
 684
 685        self._record_stage_success("batch_detect_objects")
 686        return self.object_detection_df

 688    def batch_process_mesh(
 689        self,
 690        pkl_name=None,
 691        object_list=None,
 692        save_data=True,
 693        phase_channel="",
 694        join_thresh=4,
 695        split_thresh=0.35,
 696        CD_width=False,
 697        smoothing=0.1,
 698        neighbor_filter_max_neighbors=None,
 699        neighbor_filter_connectivity=1,
 700    ):
 701        """
 702        Process cellular meshes from segmented masks and store the data in Cell objects.
 703
 704        After segmentation, this method can process the resulting masks and create cellular meshes from them.
 705        It first joins cells with a pole-pole distance (px) below the 'join_thresh'.
 706        Subsequently, meshes of all cells are created and cells can be split again if the constriction degree exceeds the 'split_thresh'.
 707        The data is then stored in Cell objects and can be exported in a pickle file.
 708
 709        Parameters
 710        ----------
 711        pkl_name : str, optional
 712            Name of the pickle file to save the mesh data. If None, the file will be saved as 'folder_name_meshdata.pkl'.
 713
 714        object_list : List, optional
 715            Specify a list of Image objects to be processed. If 'object_list' is None, all Image instances in 'image_objects' are processed.
 716
 717        phase_channel : str, optional
 718            Characters before the file-extension (.tif, .tiff) that the algorithm looks for when using this function (e.g., "c1", "phase").
 719
 720        join_thresh : int, optional
 721            The pole-pole distance threshold for joining cells. Cells with a distance below this threshold will have their masks joined. (default is 4)
 722
 723        split_thresh : float, optional
 724            The constriction degree threshold for splitting cells. Cells with a constriction degree exceeding this threshold will be split. (default is 0.35)
 725
 726        """
 727        self.mesh_df_collection = pd.DataFrame()
 728        self.mesh_prefilter_stats = pd.DataFrame()
 729        object_list = self._iter_processing_targets(object_list, phase_channel)
 730        mesh_frames = []
 731        prefilter_stats = []
 732
 733        for img_obj in object_list:
 734            bactoscoop_logger.info(f"PROCESSING IMAGE: {img_obj.image_name}")
 735            try:
 736                img_obj.join_split_pipeline(
 737                    join_thresh,
 738                    split_thresh,
 739                    CD_width,
 740                    smoothing,
 741                    neighbor_filter_max_neighbors=neighbor_filter_max_neighbors,
 742                    neighbor_filter_connectivity=neighbor_filter_connectivity,
 743                )
 744                mesh_df = getattr(img_obj, "processed_mesh_dataframe", None)
 745                if mesh_df is None:
 746                    raise ValueError(
 747                        f"Image '{img_obj.image_name}' did not produce a processed mesh dataframe."
 748                    )
 749                mesh_frames.append(mesh_df)
 750                if getattr(img_obj, "prefilter_summary", None) is not None:
 751                    prefilter_stats.append(
 752                        {
 753                            "image_name": img_obj.image_name,
 754                            **img_obj.prefilter_summary,
 755                        }
 756                    )
 757                img_obj.create_cell_object(verbose=False)
 758            except Exception as e:
 759                bactoscoop_logger.debug(traceback.format_exc())
 760                self._record_processing_error(
 761                    "mesh_processing", e, image_name=img_obj.image_name
 762                )
 763
 764        self.mesh_df_collection = (
 765            pd.concat(mesh_frames, ignore_index=True) if mesh_frames else pd.DataFrame()
 766        )
 767        self.mesh_prefilter_stats = (
 768            pd.DataFrame(prefilter_stats) if prefilter_stats else pd.DataFrame()
 769        )
 770
 771        if save_data:
 772            if pkl_name is not None:
 773                self._to_pickle(pkl_name, self.mesh_df_collection)
 774
 775            self._to_pickle(f"{self.name}_meshdata.pkl", self.mesh_df_collection)
 776
 777        self._record_stage_success("batch_process_mesh")

 780    def batch_load_mesh(self, pkl_name, pkl_path=None, phase_channel=""):
 781        """
 782        Load mesh data from a pickle-file and create image objects from the data.
 783
 784        Parameters
 785        ----------
 786        pkl_name : str
 787            Name of the pickle file where the mesh data is stored (e.g., 'Ecoli_meshdata.pkl').
 788
 789        pkl_path : str, optional
 790            Path to the file where the mesh data is stored. If not provided, the method will look in the image folder path.
 791
 792        phase_channel : str, optional
 793            Characters before the file-extension (.tif, .tiff) that the algorithm looks for when using this function (e.g., "c1", "phase").
 794
 795        """
 796        self.mesh_df_collection = None
 797        if pkl_path is None:
 798            pkl_path = self.image_folder_path
 799
 800        full_path = os.path.join(pkl_path, pkl_name)
 801        if not os.path.isfile(full_path):
 802            raise FileNotFoundError(f"Pickle file not found: {full_path}")
 803
 804        if full_path.lower().endswith(".parquet"):
 805            mesh_df = pd.read_parquet(full_path, engine="pyarrow")
 806            for column in ("contour", "mesh", "midline"):
 807                if column not in mesh_df:
 808                    raise ValueError(f"Mesh Parquet file is missing '{column}'.")

 810                    lambda value: np.asarray(value.tolist(), dtype=float)
 811                )
 812            self.mesh_df_collection = mesh_df
 813        else:
 814            with open(full_path, "rb") as f:
 815                self.mesh_df_collection = pickle.load(f)
 816
 817        bactoscoop_logger.info("Meshes loaded from file: %s", pkl_name)
 818        self.create_image_objects(phase_channel=phase_channel)

 965            success_percentage_list

 971    def batch_calculate_features(
 972        self,
 973        channel_method_pairs,
 974        all_data=True,
 975        reset=True,
 976        use_shifted_contours=False,
 977        shift_signal=False,
 978        max_mesh_size=600,
 979        retain_contour_on_object_mesh_failure_channels=None,
 980    ):
 981        """
 982        Calculate features of all image objects for specified channels and methods.
 983
 984        This method calculates features for all image objects based on the specified channel-method pairs. It takes a list of tuples, where each tuple contains channel names and the feature calculation method. The resulting feature dataframes are stored in a dictionary.
 985
 986        Parameters
 987        ----------
 988        channel_method_pairs : List[Tuple[List[str], str]]
 989            A list of tuples where each tuple contains the channel names and feature calculation method.
 990            Example of channel_method_pairs: [(['c1'], 'profiling'), (['c2', 'c3'], 'objects'), ([None], 'svm')].
 991            With the example pairs, this method will analyze c1 using the profiling method,
 992            then c2 and c3 using the objects method, and finally the phase contrast channel using the svm method.
 993
 994        add_profiling_data : bool, optional
 995            Whether to include profiling data in the feature calculation. (default is True)
 996
 997        reset : bool, optional
 998            If True, the dictionary storing the dataframes will be reset. If False, dataframes are added to the existing dictionary. (default is True)
 999        retain_contour_on_object_mesh_failure_channels : list[str] or None, optional
1000            Channels whose objects features retain contour-based measurements
1001            when any detected object mesh is invalid. Other channels omit that
1002            cell's objects feature row. None is equivalent to an empty list.
1003            This option does not affect other feature methods.
1004
1005        Returns
1006        -------
1007        Dict
1008            A dictionary containing dataframes with calculated features.
1009
1010        """
1011        self._require_loaded_images()
1012        if channel_method_pairs is None:
1013            raise ValueError("channel_method_pairs must be provided.")
1014        if retain_contour_on_object_mesh_failure_channels is None:
1015            retained_channels = frozenset()
1016        elif isinstance(retain_contour_on_object_mesh_failure_channels, str):
1017            raise TypeError(
1018                "retain_contour_on_object_mesh_failure_channels must be a list "
1019                "of channel names, not a string."
1020            )
1021        else:
1022            retained_channels = frozenset(
1023                retain_contour_on_object_mesh_failure_channels
1024            )
1025        if not all(isinstance(channel, str) for channel in retained_channels):
1026            raise TypeError(
1027                "retain_contour_on_object_mesh_failure_channels must contain "
1028                "only channel names."
1029            )
1030
1031        self.merged_features = None
1032        if reset:
1033            self.feature_dataframes = {}
1034
1035        for channels, method in channel_method_pairs:
1036            normalized_channels = (
1037                [None]
1038                if channels == [None]
1039                else self._normalize_channel_list(channels, argument_name="channels")
1040            )
1041            for channel in normalized_channels:
1042                key = f"{method}_{channel}_features"
1043                self.feature_dataframes.pop(key, None)
1044                bactoscoop_logger.info(
1045                    f"PROCESSING CHANNEL: {channel}, method: {method}"
1046                )
1047                feature_dfs = []
1048
1049                for img_obj in tqdm_if_verbose(self.image_objects):
1050                    if channel is not None and (
1051                        img_obj.channels is None or channel not in img_obj.channels
1052                    ):
1053                        self._record_processing_error(
1054                            "image_feature_calculation",
1055                            ValueError(
1056                                f"Channel {channel} is not loaded for image '{img_obj.image_name}'."
1057                            ),
1058                            image_name=img_obj.image_name,
1059                            channel=channel,
1060                        )
1061                        continue
1062                    try:
1063                        result_df = img_obj.calculate_features(
1064                            method,
1065                            channel,
1066                            all_data,
1067                            use_shifted_contours,
1068                            shift_signal,
1069                            max_mesh_size,
1070                            retained_channels,
1071                        )
1072                    except Exception as e:
1073                        bactoscoop_logger.debug(traceback.format_exc())
1074                        self._record_processing_error(
1075                            "image_feature_calculation",
1076                            e,
1077                            image_name=img_obj.image_name,
1078                            channel=channel,
1079                        )
1080                        continue
1081                    if result_df is not None:
1082                        feature_dfs.append(result_df)
1083                    for feature_error in getattr(img_obj, "feature_errors", []):
1084                        self._record_cell_error(
1085                            "cell_feature_calculation",
1086                            RuntimeError(feature_error["error"]),
1087                            image_name=img_obj.image_name,
1088                            channel=channel,
1089                            cell_id=feature_error.get("cell_id"),
1090                            error_type=feature_error.get("error_type"),
1091                        )
1092
1093                if feature_dfs:
1094                    concatenated_df = pd.concat(feature_dfs, axis=0)
1095                    concatenated_df.reset_index(drop=True, inplace=True)
1096                    self.feature_dataframes[key] = concatenated_df
1097                    has_rows = not concatenated_df.empty
1098                else:
1099                    has_rows = False
1100                if not has_rows:
1101                    self._record_processing_error(
1102                        "batch_calculate_features",
1103                        ValueError(
1104                            f"No feature rows were produced for channel {channel}, method {method}."
1105                        ),
1106                        channel=channel,
1107                    )
1108
1109        self._record_stage_success("batch_calculate_features")
1110        return self.feature_dataframes

1143    def _apply_curation_labels(self):
1144        """Apply curation labels by stable image identity, with frame fallback."""
1145        image_objects_by_name = {}
1146        image_objects_by_frame = {}
1147        for image in self.image_objects:
1148            image_objects_by_name.setdefault(image.image_name, []).append(image)
1149            image_objects_by_frame.setdefault(image.frame, []).append(image)
1150
1151        for _, row in self.curated_df.iterrows():
1152            image_name = row.get("image_name")
1153            cell_id = row["cell_id"]
1154            label = row["label"]
1155            image = None
1156
1157            if pd.notna(image_name):
1158                matches = image_objects_by_name.get(image_name, [])
1159                if len(matches) == 1:
1160                    image = matches[0]
1161                else:
1162                    reason = (
1163                        f"Image name '{image_name}' is not unique."
1164                        if len(matches) > 1
1165                        else f"No image object matches image name '{image_name}'."
1166                    )
1167                    self._record_processing_error(
1168                        "curation_mapping",
1169                        ValueError(reason),
1170                        image_name=image_name,
1171                        cell_id=cell_id,
1172                    )
1173                    continue
1174
1175            if image is None:
1176                frame = row.get("frame")
1177                matches = image_objects_by_frame.get(frame, [])
1178                if len(matches) == 1:
1179                    image = matches[0]
1180                else:
1181                    self._record_processing_error(
1182                        "curation_mapping",
1183                        ValueError(
1184                            f"Could not uniquely map curated cell {cell_id} to image "
1185                            f"(image_name={image_name!r}, frame={frame!r})."
1186                        ),
1187                        image_name=image_name if pd.notna(image_name) else None,
1188                        cell_id=cell_id,
1189                    )
1190                    continue
1191
1192            frame = row.get("frame")
1193            if pd.notna(frame) and image.frame != frame:
1194                self._record_processing_error(
1195                    "curation_mapping",
1196                    ValueError(
1197                        f"Curated cell {cell_id} names image '{image.image_name}' "
1198                        f"but has frame {frame}; expected {image.frame}."
1199                    ),
1200                    image_name=image.image_name,
1201                    cell_id=cell_id,
1202                )
1203                continue
1204
1205            cells = [cell for cell in image.cells if cell.cell_id == cell_id]
1206            if len(cells) != 1:
1207                reason = (
1208                    f"Cell ID {cell_id} was not found in image '{image.image_name}'."
1209                    if not cells
1210                    else f"Cell ID {cell_id} is not unique in image '{image.image_name}'."
1211                )
1212                self._record_processing_error(
1213                    "curation_mapping",
1214                    ValueError(reason),
1215                    image_name=image.image_name,
1216                    cell_id=cell_id,
1217                )
1218                continue
1219            if label == 0:
1220                image.cells.remove(cells[0])
1221
1222        self._record_stage_success("curate_dataset")

1224    def curate_dataset(
1225        self, path_to_model, cols=4, control=False, save_curated_data=True
1226    ):
1227        """
1228        Curate the dataset based on a trained support vector machine model.
1229
1230        This method calculates SVM features and uses a trained SVM model to classify cells based on these features.
1231        It curates the dataset by discarding cells that are classified as label 0 (non-interesting).
1232        Optionally, it can include positive and negative control cells.
1233
1234        Parameters:
1235            path_to_model (str): Path to the trained SVM model.
1236
1237            cols (int): Retained for backward compatibility; currently unused.
1238
1239            control (bool): Whether to include positive and negative control cells.
1240
1241            save_curated_data (bool): Whether to write the curated mesh pickle.
1242                The in-memory mesh dataframe is updated either way.
1243
1244        Returns:
1245            None
1246        """

1320    def export_curated_masks(self, output_subfolder="curated_masks", overwrite=True):
1321        """Write one label mask per image from the SVM-curated cell contours.
1322
1323        This method is intended to be run after :meth:`curate_dataset`. It never
1324        changes the original Omnipose masks in ``masks``. Instead, it writes new
1325        masks to ``output_subfolder`` and records how every curated label overlaps
1326        the original labels, so the output can be used directly by BactoFeatures.
1327
1328        Parameters
1329        ----------
1330        output_subfolder : str, optional
1331            Folder below ``image_folder_path`` for the exported label TIFFs.
1332        overwrite : bool, optional
1333            If ``False``, fail before replacing an existing curated-mask TIFF.
1334
1335        Returns
1336        -------
1337        tuple[list[str], pandas.DataFrame]
1338            Paths to the exported masks and a provenance table with one row per
1339            curated cell.
1340        """
1341        self._require_image_folder_path()
1342        if not self.image_objects:
1343            raise ValueError(
1344                "No image objects are available. Load curated mesh data or run "
1345                "curate_dataset() before exporting curated masks."
1346            )
1347
1348        output_dir = os.path.join(self.image_folder_path, output_subfolder)
1349        os.makedirs(output_dir, exist_ok=True)
1350        mask_paths = []
1351        provenance_rows = []
1352
1353        for image in self.image_objects:
1354            if image.mask is None:
1355                raise ValueError(f"Original mask is not loaded for {image.image_name}.")
1356
1357            max_label = len(image.cells)
1358            output_dtype = np.uint16 if max_label <= np.iinfo(np.uint16).max else np.uint32
1359            curated_mask = np.zeros(image.mask.shape, dtype=output_dtype)
1360            stem = os.path.splitext(image.image_name)[0]
1361            output_path = os.path.join(output_dir, f"{stem}_curated_masks.tif")
1362            if os.path.exists(output_path) and not overwrite:
1363                raise FileExistsError(
1364                    f"Curated mask already exists: {output_path}. Set overwrite=True to replace it."
1365                )
1366
1367            for curated_label, cell in enumerate(image.cells, start=1):
1368                contour = np.asarray(cell.contour, dtype=float)
1369                if contour.ndim != 2 or contour.shape[1] != 2 or len(contour) < 3:
1370                    bactoscoop_logger.warning(
1371                        "Skipping curated cell %s in %s: invalid contour.",
1372                        cell.cell_id,
1373                        image.image_name,
1374                    )
1375                    continue
1376
1377                cell_mask = np.zeros(image.mask.shape, dtype=np.uint8)
1378                # Cell contours are (row, column); OpenCV expects (x, y).
1379                polygon = np.rint(contour[:, ::-1]).astype(np.int32).reshape(-1, 1, 2)
1380                cv.fillPoly(cell_mask, [polygon], color=1)
1381                cell_pixels = cell_mask.astype(bool)
1382                if not cell_pixels.any():
1383                    bactoscoop_logger.warning(
1384                        "Skipping curated cell %s in %s: contour has no in-image pixels.",
1385                        cell.cell_id,
1386                        image.image_name,
1387                    )
1388                    continue
1389
1390                curated_mask[cell_pixels] = curated_label
1391                source_labels = np.unique(image.mask[cell_pixels])
1392                source_labels = source_labels[source_labels != 0].astype(int).tolist()
1393                rows, columns = np.nonzero(cell_pixels)
1394                provenance_rows.append(
1395                    {
1396                        "image_name": image.image_name,
1397                        "frame": image.frame,
1398                        "curated_label": curated_label,
1399                        "bactoscoop_cell_id": cell.cell_id,
1400                        "centroid_x_px": float(columns.mean()),
1401                        "centroid_y_px": float(rows.mean()),
1402                        "area_px": int(cell_pixels.sum()),
1403                        "original_mask_path": os.path.join(
1404                            self.image_folder_path, "masks", f"{stem}_cp_masks.tif"
1405                        ),
1406                        "original_mask_labels": ";".join(map(str, source_labels)),
1407                        "curated_mask_path": output_path,
1408                    }
1409                )
1410
1411            tifffile.imwrite(output_path, curated_mask)
1412            mask_paths.append(output_path)
1413
1414        provenance = pd.DataFrame(provenance_rows)
1415        provenance_path = os.path.join(output_dir, "curated_mask_provenance.csv")
1416        provenance.to_csv(provenance_path, index=False)
1417        bactoscoop_logger.info(
1418            "Exported %s curated masks and %s curated cells to %s",
1419            len(mask_paths),
1420            len(provenance),
1421            output_dir,
1422        )
1423        return mask_paths, provenance

1425    def merge_dataframes(self, include_metadata_tag=False, discard_morphological_nan=False):
1426        """
1427        Merge feature dataframes into a single dataframe.
1428    
1429        Merges multiple feature dataframes into a single dataframe, ensuring that columns are properly renamed
1430        to identify the channel and removes duplicates based on specific columns.
1431    
1432        Args:
1433            include_metadata_tag (bool): Whether to rename 'image_name', 'cell_id', and 'frame' to 'Metadata_*'.
1434            discard_nan (bool): Whether to discard rows with NaN in 'cell_area' and 'cell_length'.
1435    
1436        Returns:
1437            pd.DataFrame: The merged feature dataframe.
1438        """
1439        if not self.feature_dataframes:
1440            raise ValueError("No feature dataframes are available to merge.")
1441
1442        merged_dataframes = []
1443
1444        for key, df in self.feature_dataframes.items():
1445            # Keys are "method_channel_features"; channel names may contain underscores.
1446            if not key.endswith("_features"):
1447                raise ValueError(f"Invalid feature dataframe key: {key!r}")
1448            method_and_channel = key[: -len("_features")]
1449            method, separator, channel = method_and_channel.partition("_")
1450            if not separator or not method or not channel:
1451                raise ValueError(f"Invalid feature dataframe key: {key!r}")
1452
1453            # Set a prefix to identify the channel in column names if the channel is not None
1454            if channel != "None":
1455                df = df.rename(
1456                    columns={
1457                        col: f"{channel}_{col}"
1458                        for col in df.columns
1459                        if col not in ["image_name", "cell_id", "frame"]
1460                    }
1461                )
1462
1463            # Append the modified dataframe to the list
1464            merged_dataframes.append(df)
1465
1466        # Concatenate the dataframes vertically
1467        merged_data = pd.concat(merged_dataframes)
1468
1469        # Merge the dataframes based on 'image_name', 'cell_id', and 'frame'
1470        result = merged_data.pivot_table(
1471            index=["image_name", "cell_id", "frame"], aggfunc="first"
1472        ).reset_index()
1473
1474        # Discard rows with NaN in 'cell_area' and 'cell_length' if discard_nan is True
1475        if discard_morphological_nan:
1476            result = result.dropna(subset=["cell_area", "cell_length"])
1477
1478        if include_metadata_tag:
1479            # Rename 'image_name', 'cell_id', and 'frame' to 'Metadata_*'
1480            result = result.rename(
1481                columns={
1482                    "image_name": "Metadata_image_name",
1483                    "cell_id": "Metadata_cell_id",
1484                    "frame": "Metadata_frame"
1485                }
1486            )
1487
1488        # Reset the index to make it look like the final result
1489        self.merged_features = result.reset_index(drop=True)
1490
1491        return self.merged_features

1529    def _atomic_write(self, filename, writer):
1530        self._require_image_folder_path()
1531        if os.path.basename(filename) != filename or filename in {"", ".", ".."}:
1532            raise ValueError("Output filename must be a name within the image folder.")
1533        file_path = os.path.join(self.image_folder_path, filename)
1534        descriptor, temporary_path = tempfile.mkstemp(
1535            prefix=f".{filename}.", suffix=".tmp", dir=self.image_folder_path
1536        )
1537        os.close(descriptor)
1538        try:
1539            writer(temporary_path)
1540            with open(temporary_path, "rb+") as temporary_file:
1541                os.fsync(temporary_file.fileno())
1542            os.replace(temporary_path, file_path)
1543        finally:
1544            if os.path.exists(temporary_path):
1545                os.unlink(temporary_path)

1547    def _to_parquet(self, filename, data, mesh_data=False):
1548        if not filename.lower().endswith(".parquet"):
1549            raise ValueError("Parquet filename must end in .parquet")
1550        output = data.copy()
1551        if mesh_data:
1552            for column in ("contour", "mesh", "midline"):
1553                if column not in output:
1554                    raise ValueError(f"Mesh dataframe is missing '{column}'.")
1555                output[column] = output[column].map(
1556                    lambda value: np.asarray(value).tolist()
1557                )
1558        self._atomic_write(
1559            filename,
1560            lambda path: output.to_parquet(
1561                path, engine="pyarrow", compression="zstd", index=False
1562            ),
1563        )
